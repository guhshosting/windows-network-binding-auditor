cmake_minimum_required(VERSION 3.24)
project(windows_network_binding_auditor VERSION 1.0.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
 message(FATAL_ERROR "Windows and MSVC required")
endif()
option(WNBA_ANALYZE "MSVC analysis" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/W4 /WX /permissive- /utf-8 /sdl /guard:cf)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
if(WNBA_ANALYZE)
 add_compile_options(/analyze)
endif()
add_link_options(/DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /guard:cf)
add_library(audit_core src/model.cpp)
target_include_directories(audit_core PUBLIC src)
add_executable(windows-network-binding-auditor src/main.cpp src/windows.cpp)
target_link_libraries(windows-network-binding-auditor PRIVATE audit_core iphlpapi ole32 uuid)
add_executable(audit_tests tests/tests.cpp)
target_link_libraries(audit_tests PRIVATE audit_core)
enable_testing()
add_test(NAME fixtures COMMAND audit_tests)
add_test(NAME version COMMAND windows-network-binding-auditor --version)
add_test(NAME help COMMAND windows-network-binding-auditor --help)
install(TARGETS windows-network-binding-auditor RUNTIME DESTINATION .)
